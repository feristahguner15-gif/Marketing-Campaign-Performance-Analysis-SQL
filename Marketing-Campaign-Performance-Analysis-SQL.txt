with facebook_ads as (
select
fc. campaign_name,
fa. adset_name,
SUM(fbd.spend) as spend,
SUM(fbd. value) as value
from facebook_ads_basic_daily fbd
left join facebook_adset fa  on fa. adset_id = fbd. adset_id
left join facebook_campaign fc on fc. campaign_id = fbd. campaign_id
group by fc. campaign_name, fa. adset_name
),
google_ads as(
select
campaign_name,
adset_name,
SUM(spend) as spend,
SUM(value) as value
from google_ads_basic_daily
group by campaign_name, adset_name
),
all_ads as (
select 'facebook_ads' as media_source,*
from facebook_ads
UNION all
select 'google_ads' as media_source, * 
from google_ads
)
select 
media_source,
campaign_name,
adset_name,SUM(spend) as toplam_maliyet,
(SUM(value)::numeric - SUM(spend):: numeric /SUM(spend):: numeric * 100) as ROMI
from all_ads
where value is not null and spend <> 0
group by media_source, campaign_name, adset_name 
having SUM(spend) > 500000
order by ROMI desc;
